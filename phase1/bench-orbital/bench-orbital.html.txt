// ---------- classification (what the hub would publish). Every row lands in exactly one class.
function classify(csv, status) {
  const L = csv.trim().split(/\r?\n/); const h = L[0].split(','); const ix = k => h.indexOf(k);
  const iType = ix('OBJECT_TYPE'), iId = ix('NORAD_CAT_ID'), iEp = ix('EPOCH');
  const cls = new Array(L.length - 1); const c = { rows: L.length - 1, quoted: 0, by_type: {}, active: 0, inactive_payload: 0, rocket_body: 0, debris: 0,
    unknown_type: 0, payload_unmatched: 0, payload_status_unknown: 0, payload_status_decayed_conflict: 0, status_codes: {}, payload_total: 0, payload_matched: 0 };
  for (let i = 1; i < L.length; i++) {
    const line = L[i]; if (line.includes('"')) { c.quoted++; cls[i - 1] = 'skip_quoted'; continue; }
    const f = line.split(','); const t = (f[iType] || '').trim(); const id = +f[iId];
    c.by_type[t || '(blank)'] = (c.by_type[t || '(blank)'] || 0) + 1;
    let k;
    if (t === 'PAYLOAD') {
      c.payload_total++; const s = status[id];
      if (s === undefined) { k = 'payload_unmatched'; }
      else { c.payload_matched++; c.status_codes[s || '(blank)'] = (c.status_codes[s || '(blank)'] || 0) + 1;
        if (['+', 'P', 'B', 'S', 'X'].includes(s)) k = 'active';
        else if (s === '-') k = 'inactive_payload';
        else if (s === 'D') k = 'payload_status_decayed_conflict';
        else k = 'payload_status_unknown'; }      // '?', blank, anything else — never 'inactive'
    } else if (t === 'ROCKET BODY') k = 'rocket_body';
    else if (t === 'DEBRIS') k = 'debris';
    else k = 'unknown_type';
    c[k]++; cls[i - 1] = k;
  }
  return { cls, c, header: h };
}

// Space-Track CSV quotes every field. Re-emit unquoted (classifier + worker split on commas); a comma inside a field
// becomes ';' and is counted, so nothing is silently dropped. (orbital-1.1: 1.0 skipped quoted lines → 0 classified.)
function unquoteCsv(csv) {
  if (!csv.includes('"')) return { csv, fields_with_comma: 0, quoted: false };
  let o = [], line = [], f = '', q = false, commas = 0;
  for (let i = 0; i < csv.length; i++) { const ch = csv[i];
    if (q) { if (ch === '"') { if (csv[i + 1] === '"') { f += '"'; i++; } else q = false; } else if (ch === ',') { f += ';'; commas++; } else if (ch === '\n' || ch === '\r') f += ' '; else f += ch; }
    else if (ch === '"') q = true; else if (ch === ',') { line.push(f); f = ''; } else if (ch === '\n') { line.push(f); o.push(line.join(',')); line = []; f = ''; } else if (ch !== '\r') f += ch; }
  if (f || line.length) { line.push(f); o.push(line.join(',')); }
  return { csv: o.join('\n') + '\n', fields_with_comma: commas, quoted: true };
}
